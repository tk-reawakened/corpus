      #wait(a$)

      #if(a$ == "UP")

         {

         #if(select! > 1)

            #select! = select! - 1

     #if(a$ == "DOWN")

        {

           {

        }

    #if(a$ == "ENTER")

       {

       #if(select! == 1)

          {

          #done! = 1
